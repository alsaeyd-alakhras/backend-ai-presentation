Recap — Directives3 / 12
تذكير سريع — @if @foreach @forelse
@if / @foreach

@if($product->stock > 0)
  <span>متوفر</span>
@else
  <span>نفذت الكمية</span>
@endif

@foreach($products as $p)
  <li>{{ $p->name }}</li>
@endforeach
@forelse — لو القائمة فارغة

@forelse($products as $p)
  <li>{{ $p->name }}</li>
@empty
  <p>لا توجد منتجات بعد</p>
@endforelse
@forelse يوفر عليك كتابة @if(count($products) === 0) يدوياً.

compact()

.env & APP_KEY9 / 12
ملف .env — المتغيرات الأهم
APP_ENV=local
APP_DEBUG=true
APP_KEY=base64:xxxx...
APP_URL=http://localhost:8000
APP_ENV
local أثناء التطوير، production عند النشر
APP_DEBUG
true يعرض تفاصيل الأخطاء — يجب أن يكون false في production حتماً
APP_KEY
يُستخدم لتشفير الجلسات والكوكيز — بدونه Laravel لن يعمل

config() & config:cache10 / 12
دالة config() وconfig:cache
لماذا config() لا env() مباشرة؟

// ✗ لا تفعل هذا خارج ملفات config
$name = env('APP_NAME');

// ✓ اقرأ من config دائماً
$name = config('app.name');

// السبب: بعد config:cache
// ملف .env لا يُقرأ إطلاقاً
Terminal
$ php artisan config:cache
✓ Configuration cached.

# بعد أي تعديل بـ .env لازم:
$ php artisan config:clear

⚠️ نسيت تمسح config:cache بعد تعديل .env؟ التطبيق سيستمر بقراءة القيم القديمة المخزّنة!
